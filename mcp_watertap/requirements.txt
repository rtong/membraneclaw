# The environment is conda, not a venv. Reaktoro is published on conda-forge
# only — there is no PyPI wheel — so the base has to be created with a conda
# client first, and these packages layered on top with pip:
#
#   micromamba create -p <prefix>/env -c conda-forge python=3.13 reaktoro cyipopt
#   <prefix>/env/bin/pip install -r mcp_watertap/requirements.txt
#
# cyipopt also comes from conda-forge: it is the only solver reaktoro-pse is
# tested against, and it links its own ipopt in-process.
#
# Do NOT run `idaes get-extensions` against this environment. IDAES's ipopt is a
# separate subprocess build and is not needed here; on Debian 13 its libgfortran
# and liblapack requirements are satisfied by the distro packages
# (libgfortran5, liblapack3) if you do want it for the RO model alone.
watertap==1.7.0
reaktoro-pse==0.4.1

# hosted connectors depend on. Upgrading is its own piece of work with its own
